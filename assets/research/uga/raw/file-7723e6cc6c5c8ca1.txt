def finite_carrier(p):
    q=p-1;vertices=p+1
    old=['a','b','ab','ba','aba','bab','abab']
    basis=[(f'e{i}',i,i) for i in range(vertices)]
    basis += [(w,0 if w[0]=='a' else 1,1 if w[-1]=='a' else 0) for w in old]
    basis += [(f'm{w}',0,w) for w in range(2,vertices)]
    basis += [(f't{w}',w,w) for w in range(2,vertices)]
    index={n:i for i,(n,_,_) in enumerate(basis)}
    products={(index[u],index[v]):index[u+v] for u in old for v in old if u+v in old}
    A=Table(vertices,basis,products)
    assert len(A.basis)==3*p+6 and len(A.alg.radical_powers[4])==0
    return A

def local_names(A,names):
    return [[A.index[n] for n in names if A.basis[A.index[n]][1]==v] for v in range(A.vertices)]

def local_simple(A,v):
    return [[v] if i==v else [] for i in range(A.vertices)]

def combined(A,parts):
    target=ss.direct_sum(A.alg,[M for M,_,_ in parts],[1]*len(parts))
    embedded=[[] for _ in range(A.vertices)]
    for _,local,embedding in parts:
        for v,indices in enumerate(local):embedded[v].extend(embedding(b) for b in indices)
    return target,embedded

def kernel_iso(M,target,embedded,label):
    K,info,certificate=cover(M);lin=info['linear_data']
    copies={v:c for c,(v,_) in enumerate(lin['copies'])}
    assert len(copies)==len(lin['copies'])
    maps=[]
    for v,part in enumerate(embedded):
        pos={key:j for j,key in enumerate(lin['bases'][v])}
        cols=[]
        for label_v,b in part:
            key=(copies[label_v],None if b<A.vertices else b-A.vertices)
            cols.append(lin['readers'][v](rk.unit_vector(len(pos),pos[key])))
        maps.append(cols)
    checked=ss.check_isomorphism(target,K,ss.encode(maps))
    for cols,d in zip(maps,target.dimensions):unit_basis(cols,d)
    counts['actual_kernel_isomorphisms']+=1
    return {'label':label,'cover':certificate,'target':ss.module_payload(target),'isomorphism':checked}

def actual_recurrent_matrix(A,w):
    # Q=Pv + Pf + Pw, with columns literally x |-> x b, x m_w, x t_w.
    labels=[1,0,w];locals_by_copy=[projective(A,v)[1] for v in labels]
    basis=[[(copy,b) for copy,local in enumerate(locals_by_copy) for b in local[v]] for v in range(A.vertices)]
    pos=[{key:j for j,key in enumerate(part)} for part in basis]
    entries=[(1,A.index['b']),(2,A.index[f'm{w}']),(2,A.index[f't{w}'])]
    maps=[]
    for v,part in enumerate(basis):
        columns=[]
        for copy,b in part:
            target,entry=entries[copy];value=A.mult(b,entry)
            col=[F(0)]*len(part)
            if value is not None:col[pos[v][target,value]]=F(1)
            columns.append(col)
        maps.append(columns)
    for v,cols in enumerate(maps):
        assert all(not any(rk.combine(cols,col,len(cols))) for col in cols)
        chosen,_=rk.independent(cols);image=[cols[i] for i in chosen]
        kernel=rk.nullspace(cols,len(cols));reader=sr.coordinates_reader(kernel,len(cols))
        image_in_kernel=[reader(col) for col in image]
        unit_basis(image_in_kernel,len(kernel))
        unit_basis(kernel+[rk.unit_vector(len(cols),i) for i in chosen],len(cols))
    for r in A.radical:
        _,target,source=A.basis[r]
        actions=[]
        for copy,b in basis[source]:
            col=[F(0)]*len(basis[target]);value=A.mult(r,b)
            if value is not None:col[pos[target][copy,value]]=F(1)
            actions.append(col)
        for i,col in enumerate(maps[source]):
            lhs=rk.combine(actions,col,len(basis[target]))
            rhs=rk.combine(maps[target],actions[i],len(basis[target]))
            assert lhs==rhs
    counts['literal_recurrent_differentials']+=1
    return {'labels':labels,'basis_indices':basis,'matrices':ss.encode(maps),'d_squared_zero':True,
            'image_equals_kernel_integrally':True,'A_linear':True,'entries_in_radical':True}
