# Counterexample to fixed-pair persistence => reverse visibility.
T = Table(2,[('e0',0,0),('e1',1,1),('a',1,0),('x',1,1)],{})

# Literal End_D(D/t + D/t^2 + D/t^3), without identifying homology
# with a graph or with an infinite Anick resolution.
hom_basis=[(i,i,0) for i in range(1,4)]
hom_basis += [(i,j,a) for i in range(1,4) for j in range(1,4)
              for a in range(max(i-j,0),i) if (i,j,a) not in hom_basis]
hom_index={v:i for i,v in enumerate(hom_basis)}
hm={}
for p,(i,j,a) in enumerate(hom_basis[3:],3):
 for q,(h,l,b) in enumerate(hom_basis[3:],3):
  if j==h and a+b<i: hm[p,q]=hom_index[i,l,a+b]

   apds.append(degree);break
  M=K
 else: raise AssertionError('finite A failed to terminate')
 acerts.append(seq)
assert apds==[2,2,1]

# Complete reduced Groebner system, in mathematical product order.
# a1:1->2, b1:2->1, a2:2->3, b2:3->2.
letters={'a1':(1,0),'b1':(0,1),'a2':(2,1),'b2':(1,2)}
rules=[(('b1','a1'),None),
       (('a1','b1'),('b2','a2')),
       (('b2','a2','a1'),None),
       (('b1','b2','a2'),None),
       (('b2','a2','b2','a2'),None)]
def nf(word):
 if word is None:return None
 word=tuple(word)
 for step in range(1000):
  for lead,replacement in rules:
   for pos in range(len(word)-len(lead)+1):
    if word[pos:pos+len(lead)]==lead:
     if replacement is None:return None
     word=word[:pos]+replacement+word[pos+len(lead):]
     break
   else:continue
   break
  else:return word
 raise AssertionError('rewrite did not terminate')
def endpoint(word):
 return letters[word[0]][0],letters[word[-1]][1]
def composable(word):
 return all(letters[word[i]][1]==letters[word[i+1]][0]
            for i in range(len(word)-1))
critical=[]
for u,ur in rules:
 for v,vr in rules:
  for overlap in range(1,min(len(u),len(v))):
   if u[-overlap:]!=v[:overlap]:continue
   word=u+v[overlap:]
   if not composable(word):continue
   left=None if ur is None else ur+v[overlap:]
   right=None if vr is None else u[:-overlap]+vr
   assert nf(left)==nf(right)
   critical.append({'word':word,'left_normal_form':nf(left),'right_normal_form':nf(right)})
for u,_ in rules:
 for v,_ in rules:
  if u!=v:assert not any(u[i:i+len(v)]==v for i in range(len(u)-len(v)+1))
normal=[]
normal_A_images=[]
arrow_h={'a1':(2,1,1),'b1':(1,2,0),'a2':(3,2,1),'b2':(2,3,0)}
for degree in range(1,6):
 for word in product(letters,repeat=degree):
  if not composable(word):continue
  n=nf(word)
  if degree==5:assert n is None
  if n==word:
   normal.append(word)
   image=hom_index[arrow_h[word[0]]]
   for name in word[1:]: image=A.mult(image,hom_index[arrow_h[name]])
   assert image is not None
   normal_A_images.append(image)
assert len(normal)==11 and len(set(normal_A_images))==11
assert set(normal_A_images)==set(range(3,14))
bbasis=[(f'e{i}',i,i) for i in range(3)]+[(' '.join(w),*endpoint(w)) for w in normal]
bindex={w:i+3 for i,w in enumerate(normal)}
bp={}
for u in normal:
 for v in normal:
  w=u+v
  if composable(w) and not any(w[i:i+len(lead)]==lead for lead,_ in rules
                    for i in range(len(w)-len(lead)+1)):
   assert w in bindex
   bp[bindex[u],bindex[v]]=bindex[w]
B=Table(3,bbasis,bp)
assert len(B.basis)==14
