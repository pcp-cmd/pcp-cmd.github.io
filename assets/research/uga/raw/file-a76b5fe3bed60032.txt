def nakayama_seven():
    words=['a','b','ab','ba','bab']
    basis=[('e0',0,0),('e1',1,1)]
    basis += [(w,0 if w[0]=='a' else 1,1 if w[-1]=='a' else 0) for w in words]
    ix={n:i for i,(n,_,_) in enumerate(basis)}
    products={(ix[u],ix[v]):ix[u+v] for u in words for v in words if u+v in words}
    return Table(2,basis,products)

def nakayama_five():
    basis=[('e0',0,0),('e1',1,1),('a',0,1),('b',1,0),('ab',0,0)]
    return Table(2,basis,{(2,3):4})

def extend_projective_column(A,label):
    # A'=[[A,Ae_label],[0,k_g]]. New LEFT radical P_g is Ae_label.
    g=A.vertices;old_n=len(A.basis)
    basis=[(f'e{i}',i,i) for i in range(g+1)]
    basis += A.basis[g:]
    column=[i for i,(_,_,r) in enumerate(A.basis) if r==label]
    basis += [(f'x{g}_{A.basis[i][0]}',A.basis[i][1],g) for i in column]
    oldmap={i:i if i<g else i+1 for i in range(old_n)}
    newmap={i:g+1+(old_n-g)+k for k,i in enumerate(column)}
    products={(oldmap[i],oldmap[j]):oldmap[k] for (i,j),k in A.products.items()}
    for i in A.radical:
        for j in column:
            value=A.mult(i,j)
            if value is not None:products[oldmap[i],newmap[j]]=newmap[value]
    result=Table(g+1,basis,products)
    return result,column,oldmap,newmap
