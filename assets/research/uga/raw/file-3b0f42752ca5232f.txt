def restricted_ext_products(A,Flabels):
    # These F resolutions terminate in degree at most three. The only
    # possible positive products are degree-one times degree-one.
    res={j:yg.Resolution(A.alg,j,3) for j in Flabels}
    output=[]
    for j in Flabels:
        for t in Flabels:
            morphisms=list(yg.cochain_maps(res[j].modules[1],t))
            for morphism in morphisms:
                lifted=yg.syzygy_map(res[j].covers[1],res[t].covers[0],morphism)
                for z in Flabels:
                    lifts,_=yg.top_reader(res[j].modules[2])
                    target_lifts,read=yg.top_reader(res[t].modules[1])
                    cols=[]
                    for lift in lifts[z]:
                        value=rk.combine(lifted[z],lift,res[t].modules[1].dimensions[z])
                        cols.append(read(z,value))
                    rank=rk.rank(cols)
                    expected=int(j==1 and t==0 and z==1)
                    assert rank==expected,(j,t,z,rank,expected)
                    output.append({'source':j,'middle':t,'target':z,'rank':rank,'comparison_values':cols})
                    counts['actual_F_Yoneda_product_checks']+=1
    return output

def graded_resolution(Y,source,expected):
    M=sr.Module.simple(Y.alg,source)
    degrees=[[0]*dim for dim in M.dimensions]
    algebra_degrees=[0,0,1,1,2]
    out=[]
    for n,expected_term in enumerate(expected):
        K,info,cert=cover(M);lin=info['linear_data'];head_degrees=[]
        for v,lift in lin['copies']:
            seen={degrees[v][i] for i,c in enumerate(lift) if c}
            assert len(seen)==1
            head_degrees.append(next(iter(seen)))
        terms=[(v,degree) for (v,_),degree in zip(lin['copies'],head_degrees)]
        assert terms==[expected_term],(source,n,terms,expected_term)
        domain_degrees=[[head_degrees[c]+(0 if b is None else algebra_degrees[Y.vertices+b])
                         for c,b in part] for part in lin['bases']]
        kernel_degrees=[]
        for v,columns in enumerate(lin['kernels']):
            local=[]
            for col in columns:
                seen={domain_degrees[v][i] for i,c in enumerate(col) if c}
                assert len(seen)==1
                local.append(next(iter(seen)))
            kernel_degrees.append(local)
        out.append({'degree':n,'projective_label_and_generator_degree':terms,
                    'kernel_coordinate_degrees':kernel_degrees,'actual_cover':cert})
        counts['graded_Yoneda_covers']+=1
        M,degrees=K,kernel_degrees
    assert not any(M.dimensions)
    return out
