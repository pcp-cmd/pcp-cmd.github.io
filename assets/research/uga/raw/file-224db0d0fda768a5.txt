# Sparse multivariate integer polynomials; no floating point or mod-p derivative.
def one(m=1):return {(0,)*m:1}
def mon(d,c=1):return {tuple(d):c} if c else {}
def add(a,b,sign=1):
 c=dict(a)
 for e,x in b.items():
  c[e]=c.get(e,0)+sign*x
  if not c[e]:del c[e]
 return c
def mul(a,b):
 c={}
 for e,x in a.items():
  for f,y in b.items():
   d=tuple(v+w for v,w in zip(e,f));c[d]=c.get(d,0)+x*y
 return {e:x for e,x in c.items() if x}
def determinant(C,m=1):
 total={}
 for p in permutations(range(len(C))):
  term=one(m)
  for i,j in enumerate(p):term=mul(term,C[i][j])
  sign=(-1)**sum(p[i]>p[j] for i in range(len(p)) for j in range(i+1,len(p)))
  total=add(total,term,sign)
 return total
def matmul(C,B):
 return [[sum_poly([mul(C[i][k],B[k][j]) for k in range(len(C))])
           for j in range(len(C))] for i in range(len(C))]
def sum_poly(parts):
 out={}
 for p in parts:out=add(out,p)
 return out
def diff(p):return {(e[0]-1,):e[0]*a for e,a in p.items() if e[0]}
def encoded(p):return [{'exponents':list(e),'coefficient':a} for e,a in sorted(p.items())]
def cartan(A,degrees):
 C=[[{} for _ in range(A.vertices)] for _ in range(A.vertices)]
 for (_,l,r),d in zip(A.basis,degrees):C[l][r]=add(C[l][r],mon((d,)))
 return C
def log_coeffs(D,last):
 assert D.get((0,),0)==1
 d={e[0]:a for e,a in D.items()};u=max(d);g=[]
 for i in range(last+1):
  value=(i+1)*d.get(i+1,0)
  value-=sum(d.get(j,0)*g[i-j] for j in range(1,min(i,u)+1))
  g.append(value)
 assert all(type(v) is int for v in g)
 return g

def homogeneous(vector,degrees):

def check_identity(C,B,last):
 E=[[{} for _ in row] for row in C]
 trace={}
 for n,Bn in enumerate(B):
  product=matmul(C,Bn)
  for i in range(len(C)):
   for j in range(len(C)):E[i][j]=add(E[i][j],product[i][j],(-1)**n)
  for x in range(len(C)):
   for y in range(len(C)):
    trace=add(trace,mul(diff(C[y][x]),Bn[x][y]),(-1)**n)
 for i,row in enumerate(E):
  for j,p in enumerate(row):
   trunc={e:a for e,a in p.items() if e[0]<=last}
   assert trunc==(one() if i==j else {})
   extra['integer_Euler_cells']+=last+1

 assert [trace.get((d,),0) for d in range(last+1)]==g
 extra['log_trace_coefficients']+=last+1
 return D,g
