def cube_layers(T):
    W=sorted(set(T.products.values()))
    V=[b for b in T.radical if b not in W]
    assert all(T.mult(w,r) is None and T.mult(r,w) is None for w in W for r in T.radical)
    assert set(T.products.values())=={T.mult(a,b) for a,b in product(V,repeat=2) if T.mult(a,b) is not None}
    return V,W

def bar_linear_cells(T,V,n):
    """Actual kernel in bar degree n, internal degree n. No boundaries there."""
    if n==0:
        return [[int(i==j) for j in range(T.vertices)] for i in range(T.vertices)],0
    words=[(b,) for b in V]
    for _ in range(n-1):
        words=[word+(b,) for word in words for b in V if T.basis[word[-1]][2]==T.basis[b][1]]
    cells=[[0]*T.vertices for _ in range(T.vertices)]
    for source,target in product(range(T.vertices),repeat=2):
        local=[w for w in words if T.basis[w[-1]][2]==source and T.basis[w[0]][1]==target]
        outputs=[]
        for word in local:
            col={}
            for p in range(n-1):
                value=T.mult(word[p],word[p+1])
                if value is not None:
                    out=word[:p]+(value,)+word[p+2:]
                    col[out]=col.get(out,F(0))+F((-1)**(p+1))
            outputs.append(col)
        rows=sorted({row for col in outputs for row in col})
        columns=[[col.get(row,F(0)) for row in rows] for col in outputs]
        cells[source][target]=len(local)-rk.rank(columns)
    return cells,len(words)

def map_composition(first,second,target):
    return [[rk.combine(second[v],col,target.dimensions[v]) for col in first[v]]
            for v in range(target.algebra.vertices)]

def degree_one_products(T,max_degree):
    """Only products of actual Ext-one classes, with checked ordinary lifts."""
    resolutions=[yg.Resolution(T.alg,j,max_degree+1) for j in range(T.vertices)]
    current={}
    comparisons=0
    for source,target in product(range(T.vertices),repeat=2):
        current[source,target]=list(yg.cochain_maps(resolutions[source].modules[1],target))
    result=[]
    for n in range(1,max_degree+1):
        cells=[[0]*T.vertices for _ in range(T.vertices)]
        reduced={}
        for source,target in product(range(T.vertices),repeat=2):
            M=resolutions[source].modules[n]
            lifts,_=yg.top_reader(M)
            vectors=[[rk.combine(phi[target],lift,1)[0] for lift in lifts[target]]
                     for phi in current[source,target]]
            selected,_=rk.independent(vectors)
            cells[source][target]=len(selected)
            reduced[source,target]=[current[source,target][i] for i in selected]
        result.append(cells)
        if n==max_degree:break
        nxt={(i,j):[] for i,j in product(range(T.vertices),repeat=2)}
        for source,middle in product(range(T.vertices),repeat=2):
            for phi in reduced[source,middle]:
                lift=yg.syzygy_map(resolutions[source].covers[n],resolutions[middle].covers[0],phi)
                comparisons+=1
                for target in range(T.vertices):
                    for psi in yg.cochain_maps(resolutions[middle].modules[1],target):
                        composed=map_composition(lift,psi,sr.Module.simple(T.alg,target))
                        yg.check_module_map(resolutions[source].modules[n+1],
                                            sr.Module.simple(T.alg,target),composed)
                        nxt[source,target].append(composed)
        current=nxt
    return result,comparisons

def socle_dimensions(M):
    result=[]
    for vertex in range(M.algebra.vertices):
        height=sum(M.dimensions[b['left']] for b in M.algebra.basis if b['right']==vertex)
        columns=[]
        for j in range(M.dimensions[vertex]):
            col=[]
            for b,action in zip(M.algebra.basis,M.actions):
                if b['right']==vertex:col.extend(action[j])
            assert len(col)==height
            columns.append(col)
        result.append(M.dimensions[vertex]-rk.rank(columns))
    return result
