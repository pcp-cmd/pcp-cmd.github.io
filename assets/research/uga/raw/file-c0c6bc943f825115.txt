def tensor_table(C,D):
    m=D.vertices
    pairs=[(i,j) for i in range(C.vertices) for j in range(D.vertices)]
    pairs += [(i,j) for i in range(len(C.basis)) for j in range(len(D.basis))
              if i>=C.vertices or j>=D.vertices]
    ix={pair:k for k,pair in enumerate(pairs)}
    basis=[(f'{C.basis[i][0]}:{D.basis[j][0]}',
            C.basis[i][1]*m+D.basis[j][1],C.basis[i][2]*m+D.basis[j][2]) for i,j in pairs]
    vertices=C.vertices*D.vertices
    rad_products={}
    for x in range(vertices,len(pairs)):
        i,j=pairs[x]
        for y in range(vertices,len(pairs)):
            k,l=pairs[y];c,d=C.mult(i,k),D.mult(j,l)
            if c is not None and d is not None:rad_products[x,y]=ix[c,d]
    A=Table(vertices,basis,rad_products)
    def peirce(T,u,v):return sum(l==u and r==v for _,l,r in T.basis)
    for y in range(A.vertices):
        for z in range(A.vertices):
            assert peirce(A,y,z)==peirce(C,y//m,z//m)*peirce(D,y%m,z%m)
    return A,{'tensor_basis_pairs':pairs,'Peirce_matrix':[[peirce(A,y,z) for z in range(A.vertices)] for y in range(A.vertices)]}

def factor_terms(side,label,n):
    if side=='left':
        return [label] if n==0 else ([1] if label==0 and n==1 else ([] if label==0 else [0]))
    return [label] if n==0 else ([0] if label==1 and n==1 else ([] if label==1 else [1]))

def tensor_terms(side,labels,n):
    terms=[0]*4
    for i in range(n+1):
        for x in factor_terms(side,labels[0],i):
            for y in factor_terms(side,labels[1],n-i):terms[x*2+y]+=1
    return terms
