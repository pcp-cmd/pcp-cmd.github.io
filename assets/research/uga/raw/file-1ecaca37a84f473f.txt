def power_indices(T,k):
    columns=T.alg.radical_powers[k-1] if k-1<len(T.alg.radical_powers) else []
    selected=[]
    for col in columns:
        support=[i for i,x in enumerate(col) if x]
        assert len(support)==1 and col[support[0]]==1
        selected.append(T.vertices+support[0])
    return selected

def local_indices(T,indices):
    return [[b for b in indices if T.basis[b][1]==v] for v in range(T.vertices)]

def top_matrix(T,k):
    gens=sorted(set(power_indices(T,k))-set(power_indices(T,k+1)))
    matrix=[[0]*T.vertices for _ in range(T.vertices)]
    for b in gens:
        _,y,z=T.basis[b]
        matrix[y][z]+=1
    return gens,matrix

def matmul(X,Y):
    n=len(X)
    return [[sum(X[i][k]*Y[k][j] for k in range(n)) for j in range(n)] for i in range(n)]

def literal_corner(T,labels):
    names=list(labels)
    ren={v:j for j,v in enumerate(names)}
    keep=names+[b for b in T.radical if T.basis[b][1] in ren and T.basis[b][2] in ren]
    remap={b:j for j,b in enumerate(keep)}
    bs=[(T.basis[b][0],ren[T.basis[b][1]],ren[T.basis[b][2]]) for b in keep]
    mul={(remap[i],remap[j]):remap[k] for (i,j),k in T.products.items()
         if i in remap and j in remap}
    return Table(len(labels),bs,mul),keep

def projective_ideal(T,k):
    inds=power_indices(T,k)
    M=supported_module(T,local_indices(T,inds))
    K,info,cert=cover(M)
    assert not any(K.dimensions)
    gens,top=top_matrix(T,k)
    assert [sum(row) for row in top]==info['cover_multiplicities']
    # The full cover is an isomorphism; all chosen projective images are retained.
    for v in range(T.vertices):
        unit_basis(info['linear_data']['images'][v],M.dimensions[v])
    return inds,gens,top,cert
