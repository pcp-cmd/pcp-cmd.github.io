def actual_corner(original,retained):
    labelmap={v:i for i,v in enumerate(retained)}
    selected=[v for v in retained]
    selected += [b for b in original.radical if original.basis[b][1] in labelmap and original.basis[b][2] in labelmap]
    ix={b:i for i,b in enumerate(selected)}
    basis=[(original.basis[b][0],labelmap[original.basis[b][1]],labelmap[original.basis[b][2]]) for b in selected]
    products={}
    for i in range(len(retained),len(selected)):
        for j in range(len(retained),len(selected)):
            value=original.mult(selected[i],selected[j])
            if value is not None:
                assert value in ix
                products[i,j]=ix[value]
    C=Table(len(retained),basis,products)
    return C,labelmap,selected

def finite_resolution(A,source,pd_expected):
    M=sr.Module.simple(A.alg,source);out=[]
    for n in range(pd_expected+1):
        K,info,cert=cover(M)
        assert any(info['cover_multiplicities'])
        out.append({'degree':n,'actual_cover':cert})
        M=K
    assert not any(M.dimensions),(source,pd_expected,M.dimensions)
    return out
