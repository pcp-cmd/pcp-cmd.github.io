def loewy(M):
 current=[[rk.unit_vector(d,i) for i in range(d)] for d in M.dimensions]
 for step in range(1,12):
  cols=[[] for _ in M.dimensions]
  for a,actions in zip(M.algebra.basis,M.actions):
   cols[a['left']]+=[rk.combine(actions,v,M.dimensions[a['left']]) for v in current[a['right']]]
  nxt=[]
  for part in cols:
   chosen,_=rk.independent(part);nxt.append([part[i] for i in chosen])
  if not any(nxt):return step
  current=nxt
 raise AssertionError('Loewy calibration did not terminate')
