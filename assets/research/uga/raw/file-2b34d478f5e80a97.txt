def nakayama(c):
    N=len(c)
    basis=[(f'e{i}',i,i) for i in range(N)]
    paths={(i,0):i for i in range(N)}
    for i in range(N):
        for l in range(1,c[i]):
            paths[i,l]=len(basis)
            basis.append((f'p{i}_{l}',(i+l)%N,i))
    products={}
    for (i,l),b in paths.items():
        if not l:continue
        for (j,m),a in paths.items():
            if m and j==(i+l)%N and l+m<c[i]:
                products[a,b]=paths[i,l+m]
    return Table(N,basis,products),paths
