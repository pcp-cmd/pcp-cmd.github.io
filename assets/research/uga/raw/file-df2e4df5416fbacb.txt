     ('t',1,1),('v',0,2),('w',0,1)]
basis=[(f'e{i}',i,i) for i in range(6)]
basis.extend([(f'D_{name}',r,l) for name,l,r in old])

# W=LEFT L_B2 tensor RIGHT (e_D2 D / <D_w>).

ix={name:i for i,(name,_,_) in enumerate(basis)}
products={}
for x,y,z in [('a','b','t'),('d','s','v'),('d','c','w')]:
    products[ix[f'B_{x}'],ix[f'B_{y}']]=ix[f'B_{z}']
    products[ix[f'D_{y}'],ix[f'D_{x}']]=ix[f'D_{z}']
for x,y,z in [('z0','D_b','zb'),('z0','D_c','zc'),('z0','D_t','zt'),('zb','D_a','zt')]:

def loc(T,names):
    return [[T.index[n] for n in names if T.basis[T.index[n]][1]==v] for v in range(T.vertices)]

def checked_kernel(T,M,target,embedded,label):
    K,info,cert=cover(M)
    lin=info['linear_data']
    assert len(lin['copies'])==1
    head=lin['copies'][0][0]
    maps=[]
    for v,part in enumerate(embedded):
        pos={head if r is None else T.vertices+r:j for j,(copy,r) in enumerate(lin['bases'][v])}
        maps.append([lin['readers'][v](rk.unit_vector(info['cover_dimensions'][v],pos[b])) for b in part])
    iso=ss.check_isomorphism(target,K,ss.encode(maps))
    for cols,d in zip(maps,K.dimensions):
        unit_basis(cols,d)
    counts['actual_kernel_isomorphisms']+=1
    return {'label':label,'actual_cover':cert,'target':ss.module_payload(target),'isomorphism':iso}
