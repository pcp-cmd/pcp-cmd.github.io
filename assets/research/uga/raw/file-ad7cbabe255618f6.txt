def power_indices(k):
    columns=A.alg.radical_powers[k-1] if k-1<len(A.alg.radical_powers) else []
    selected=[]
    for col in columns:
        support=[i for i,x in enumerate(col) if x]
        assert len(support)==1 and col[support[0]]==1
        selected.append(A.vertices+support[0])
    return selected

def local_indices(indices):
    return [[b for b in indices if A.basis[b][1]==v] for v in range(A.vertices)]

def identity_iso(M,K,label):
    assert M.dimensions==K.dimensions
    checked=ss.check_isomorphism(M,K,ss.encode(
        [[rk.unit_vector(d,j) for j in range(d)] for d in M.dimensions]))
    counts['actual_kernel_isomorphisms']+=1
    return {'label':label,'isomorphism':checked}

def projective_kernel(M,origins,label):
    K,info,cert=cover(M)
    lin=info['linear_data']
    targets=[projective(A,k)[0] for _,_,k in origins]
    target=ss.direct_sum(A.alg,targets,[1]*len(targets))
    embeddings=[[] for _ in range(A.vertices)]
    for g,q,k in origins:
        copies=[j for j,(v,_) in enumerate(lin['copies']) if v==g]
        assert len(copies)==1
        copy=copies[0]
        P,local=projective(A,k)
        for v,part in enumerate(local):
            pos={(c,g if b is None else A.vertices+b):j
                 for j,(c,b) in enumerate(lin['bases'][v])}
            for b in part:
                image=A.mult(b,q)
                assert image is not None
                embeddings[v].append(lin['readers'][v](
                    rk.unit_vector(info['cover_dimensions'][v],pos[copy,image])))
    checked=ss.check_isomorphism(target,K,ss.encode(embeddings))
    for cols,d in zip(embeddings,K.dimensions):
        unit_basis(cols,d)
    counts['actual_kernel_isomorphisms']+=1
    K2,_,next_cert=cover(K)
    assert not any(K2.dimensions)
    assert any(K.dimensions)
    return {'label':label,'pd':1,'first_cover':cert,
            'FULL_kernel_projective_isomorphism':checked,'kernel_cover':next_cert}
