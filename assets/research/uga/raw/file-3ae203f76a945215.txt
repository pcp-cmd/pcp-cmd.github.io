def integral(value):
    if isinstance(value,F):return value.denominator==1
    if isinstance(value,list):return all(integral(v) for v in value)
    if isinstance(value,dict):return all(integral(v) for v in value.values())
    return isinstance(value,(int,str,bool)) or value is None

def unit_basis(columns,height):
    assert len(columns)==height and integral(columns)
    read=sr.coordinates_reader(columns,height)
    inv=[read(rk.unit_vector(height,j)) for j in range(height)]
    assert integral(inv), 'not a unimodular integer basis'
    for j in range(height):
        assert rk.combine(columns,inv[j],height)==rk.unit_vector(height,j)
    counts['unimodular_basis_certificates']+=1
    return {'basis':columns,'inverse':inv}

class Table:
    def __init__(self,vertices,basis,mult):
        self.vertices,self.basis,self.mult=vertices,basis,mult
        self.names=[b[0] for b in basis]
        self.index={n:i for i,n in enumerate(self.names)}
        self.radical=list(range(vertices,len(basis)))
        for i,j,k in product(range(len(basis)),repeat=3):
            ij,jk=mult(i,j),mult(j,k)
            assert (None if ij is None else mult(ij,k))==(None if jk is None else mult(i,jk))
            counts['full_basis_associativity']+=1
        self.payload={'vertices':vertices,'base_vertex':0,
          'radical_basis':[{'name':n,'left':l,'right':r} for n,l,r in basis[vertices:]],
          'radical_products':[]}
        for i,j in product(self.radical,repeat=2):
            out=mult(i,j)
            if out is not None:
                assert out>=vertices
                self.payload['radical_products'].append({'left':self.names[i],'right':self.names[j],
                                                        'value':{self.names[out]:1}})
        self.alg=rk.Algebra(self.payload)

    def opposite(self):
        return Table(self.vertices,[(n,r,l) for n,l,r in self.basis],lambda i,j:self.mult(j,i))

def make_B():
    modules=[(s,l) for l in [1,2,3] for s in [0,1]]
    units=[(i,i,0) for i in range(6)]
    homs=[(i,j,p) for i,(s,l) in enumerate(modules)
          for j,(t,m) in enumerate(modules) for p in range(max(0,m-l),m) if (t+p)%2==s]
    triples=units+[h for h in homs if h not in units]
    loc={h:i for i,h in enumerate(triples)}
    basis=[(f'h{i}_{j}_{p}',i,j) for i,j,p in triples]
    def mult(x,y):
        a,b,p=triples[x]; c,d,q=triples[y]
        return None if b!=c or p+q>=modules[d][1] else loc[a,d,p+q]
    return Table(6,basis,mult),modules,triples

def triangle(B,frames):
    v=B.vertices
    basis=B.basis[:v]+[('ev',v,v)]+B.basis[v:]
    B_to_A={i:i if i<v else i+1 for i in range(len(B.basis))}
    A_to_B={x:y for y,x in B_to_A.items()}
    tv=len(basis);basis.append(('tv',v,v))
    copies,from_B={},{}
    for f in frames:
        for b,(_,left,right) in enumerate(B.basis):
            if left==f:
                copies[len(basis)]=(f,b);from_B[f,b]=len(basis)
                basis.append((f'v{f}_'+B.names[b],v,right))
    def mult(i,j):
        if i in A_to_B and j in A_to_B:
            out=B.mult(A_to_B[i],A_to_B[j])
            return None if out is None else B_to_A[out]
        if i==v:return j if j==v or j==tv or j in copies else None
        if i==tv:return tv if j==v else None
        if i in copies and j in A_to_B:
            f,b=copies[i];out=B.mult(b,A_to_B[j])
            return None if out is None else from_B[f,out]
        return None
    return Table(v+1,basis,mult),B_to_A,from_B

def quotient(A,deleted):
    killed=set(deleted);old=-1
    while old!=len(killed):
        old=len(killed)
        for i in list(killed):
            for j in range(len(A.basis)):
                for out in [A.mult(i,j),A.mult(j,i)]:
                    if out is not None:killed.add(out)
    kept=[i for i in range(len(A.basis)) if i not in killed]
    labels=[i for i in range(A.vertices) if i not in deleted]
    vl={i:j for j,i in enumerate(labels)};rev={i:j for j,i in enumerate(kept)}
    basis=[(A.names[i],vl[A.basis[i][1]],vl[A.basis[i][2]]) for i in kept]
    C=Table(len(labels),basis,lambda i,j:rev.get(A.mult(kept[i],kept[j])))
    for i,j in product(range(len(A.basis)),repeat=2):
        out=A.mult(i,j)
        if i in killed or j in killed:assert out is None or out in killed
        else:assert C.mult(rev[i],rev[j])==rev.get(out)
        counts['quotient_products']+=1
    return C,labels,kept,sorted(killed)

def cover(M):
    a=M.algebra
    assert integral(M.actions)
    ss.parse_module(a,ss.module_payload(M))
    radical=M.radical_bases();top=M.top_lifts()
    top_cert=[unit_basis(radical[v]+top[v],M.dimensions[v]) for v in range(a.vertices)]
    K,info=M.syzygy(retain_maps=True);lin=info['linear_data']
    assert integral(K.actions) and integral(lin['kernels'])
    sequence_cert=[]
    for v in range(a.vertices):
        chosen,_=rk.independent(lin['images'][v])
        image_cert=unit_basis([lin['images'][v][i] for i in chosen],M.dimensions[v])
        height=info['cover_dimensions'][v]
        domain_cert=unit_basis(lin['kernels'][v]+[rk.unit_vector(height,i) for i in chosen],height)
        sequence_cert.append({'codomain':image_cert,'domain':domain_cert})
    counts['integral_cover_steps']+=1
    cert={'module':ss.module_payload(M),'kernel':ss.module_payload(K),
      'cover_multiplicities':info['cover_multiplicities'],
      'cover_dimensions':info['cover_dimensions'],
      'cover_images':lin['images'],'kernel_inclusion':lin['kernels'],
      'projective_actions':lin['projective_actions'],
      'top_basis_certificates':top_cert,'exact_sequence_basis_certificates':sequence_cert,
      'A_linear_surjective_minimal_checked':True,'all_fields_by_integral_unimodular_bases':True}
    return K,info,cert

def finite_resolution(table,vertex,expected_pd):
    M=sr.Module.simple(table.alg,vertex);certs=[]
    for n in range(expected_pd+1):
        assert sum(M.dimensions)>0
        M,_,cert=cover(M);certs.append(cert)
    assert sum(M.dimensions)==0
    return {'simple':vertex,'pd':expected_pd,'integral_cover_certificates':certs}

def projective(table,vertex):
    local=[[i for i,(_,l,r) in enumerate(table.basis) if l==v and r==vertex]
           for v in range(table.vertices)]
    pos=[{i:j for j,i in enumerate(part)} for part in local]
    dims=list(map(len,local));actions=[]
    for r in table.radical:
        _,target,source=table.basis[r];columns=[]
        for b in local[source]:
            vector=[F(0)]*dims[target];out=table.mult(r,b)
            if out is not None:vector[pos[target][out]]=F(1)
            columns.append(vector)
        actions.append(columns)
    return sr.Module(table.alg,dims,actions),local

def first_kernel_iso(table,source,target,embedded,label):
    K,info,cert=cover(sr.Module.simple(table.alg,source));lin=info['linear_data']
    maps=[]
    for v,part in enumerate(embedded):
        positions={source if r is None else table.vertices+r:j
                   for j,(copy,r) in enumerate(lin['bases'][v])}
        maps.append([lin['readers'][v](rk.unit_vector(len(positions),positions[b])) for b in part])
    checked=ss.check_isomorphism(target,K,ss.encode(maps))
    assert integral(maps)
    for columns,d in zip(maps,target.dimensions):unit_basis(columns,d)
    counts['actual_isomorphisms']+=1
    return {'label':label,'cover':cert,'target':ss.module_payload(target),'isomorphism':checked}
