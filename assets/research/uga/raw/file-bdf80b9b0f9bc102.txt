def perfect_family(p):
    vertices=p+1;old=['a','b','ab','ba','aba','bab','abab']
    basis=[(f'e{i}',i,i) for i in range(vertices)]
    basis += [(w,0 if w[0]=='a' else 1,1 if w[-1]=='a' else 0) for w in old]
    basis += [(f'm{w}',0,w) for w in range(2,vertices)]
    basis += [(f'n{w}',0,w) for w in range(2,vertices)]
    basis += [(f't{w}',w,w) for w in range(2,vertices)]
    ix={n:i for i,(n,_,_) in enumerate(basis)}
    products={(ix[u],ix[v]):ix[u+v] for u in old for v in old if u+v in old}
    products.update({(ix[f'm{w}'],ix[f't{w}']):ix[f'n{w}'] for w in range(2,vertices)})
    A=Table(vertices,basis,products)
    assert len(A.basis)==4*p+5 and len(A.alg.radical_powers[4])==0
    return A
